Setting up conky
=================


sudo apt update
sudo apt install conky-all

mkdir -p ~/.config/conky
mv ~/Downloads/YourTheme ~/.config/conky/


cd ~/.config/conky/YourTheme
conky -c conky.conf

mkdir -p ~/.local/bin
nano ~/.local/bin/conky-todo.sh

#!/bin/bash
conky -c /home/john/.config/conky/MX-ToDo/todo_sys.conf &

chmod +x ~/.local/bin/conky-todo.sh
